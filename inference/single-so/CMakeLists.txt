cmake_minimum_required(VERSION 3.16)
project(UAVLocSingleSO LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(vNavigatorInterface SHARED src/adapter.cpp src/runtime.cpp)
target_include_directories(vNavigatorInterface PUBLIC include)
target_link_libraries(vNavigatorInterface PRIVATE Threads::Threads dl)
target_compile_definitions(vNavigatorInterface PRIVATE NAV_EMBEDDED=1)
target_compile_options(vNavigatorInterface PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(vNavigatorInterface PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(NAV_EXTERNAL_PYTHON "python3" CACHE STRING "Python interpreter in the separately prepared inference environment")
add_library(adapter_external SHARED src/adapter.cpp)
target_include_directories(adapter_external PUBLIC include)
target_compile_definitions(adapter_external PRIVATE NAV_PYTHON="${NAV_EXTERNAL_PYTHON}" NAV_WORKER="${CMAKE_CURRENT_SOURCE_DIR}/python/worker.py")
target_link_libraries(adapter_external PRIVATE Threads::Threads)
add_executable(adapter_probe tests/probe.cpp)
target_link_libraries(adapter_probe PRIVATE adapter_external)
add_executable(embedded_probe tests/probe.cpp)
target_include_directories(embedded_probe PRIVATE include)
target_link_libraries(embedded_probe PRIVATE vNavigatorInterface)
add_executable(runtime_check tools/runtime_check.cpp)
target_link_libraries(runtime_check PRIVATE dl)
